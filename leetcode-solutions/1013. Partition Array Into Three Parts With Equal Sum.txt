class Solution {
public:
    bool canThreePartsEqualSum(vector<int>& arr) {

        int c=0;
        int n=arr.size();
        int sum=0;
        for(int i=0;i<n;i++){
            sum+=arr[i];
        }
        if(sum%3!=0){
            return false;
        }
        int num1=sum/3;

        int s=0;

        for(int i=0;i<n;i++){
            s+=arr[i];
            if(s==num1){
                c++;
                s=0;
            }
        }
        if(c>=3){
            return true;
        }
        else{
            return false;
        }
        

        
    }
};
