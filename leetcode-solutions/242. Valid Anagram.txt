class Solution {
public:
    bool isAnagram(string s, string t) {
        if(s.length()!=t.length()){
            return false;
        }
        int st[256]={0};

        for(int i=0;i<s.length();i++){
            st[s[i]]++;
        }

        for(int i=0;i<t.length();i++){
            st[t[i]]--;
        }

        for(int i=0;i<256;i++){
            if(st[i]!=0){
                return false;
            }
        }
        return true;
    }
};
