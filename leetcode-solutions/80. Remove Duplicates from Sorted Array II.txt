class Solution {
public:
    int removeDuplicates(vector<int>& nums) {
        vector<int>v;
        int el=nums[0];
        int count=0;
        int n=nums.size();

        for(int i=0;i<n;i++){
            if(el!=nums[i]){
                el=nums[i];
            v.push_back(nums[i]);
            count=1;
            }

            else if(el==nums[i]){
                if(count<2){
                    v.push_back(nums[i]);
                    count++;
                }
                else{
                    continue;
                }
            }

        }

        nums=v;
        return v.size();
    }
};
