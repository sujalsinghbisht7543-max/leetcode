class Solution {
public:
    bool findRotation(vector<vector<int>>& mat, vector<vector<int>>& target) {
        int n=mat.size();
        
int ans = -1;

if(mat==target){
    return true;
}
        if(ans==-1){
        for(int i=0;i<n;i++){
            for(int j=i+1;j<n;j++){
                swap(mat[i][j],mat[j][i]);
            }
        }
        for(int i=0;i<n;i++){
            int left=0;
            int right=n-1;
            while(left<right){
                swap(mat[i][left],mat[i][right]);
                left++,right--;
            }
        }
        if(mat==target){
            ans=1;
        }
        else{
            ans=-2;
        }
        }
    

        if(ans==-2){
            for(int i=0;i<n;i++){
                for(int j=i+1;j<n;j++){
                    swap(mat[i][j],mat[j][i]);
                }
            }
            for(int i=0;i<n;i++){
                int left=0;
                int right=n-1;
                while(left<right){
                    swap(mat[i][left],mat[i][right]);
                    left++,right--;
                }
            }
           
        if(mat==target){
            ans=1;
        }
        else{
            ans=-3;
        }
        }
        

        if(ans==-3){
            for(int i=0;i<n;i++){
                for(int j=i+1;j<n;j++){
                    swap(mat[i][j],mat[j][i]);
                }
            }
            for(int i=0;i<n;i++){
                int left=0;
                int right=n-1;
                while(left<right){
                    swap(mat[i][left],mat[i][right]);
                    left++,right--;
                }
            }

            if(mat==target){
                ans=1;
            }
            else{
                ans=-4;
            }

        }

if(ans>0){
    return true;
}
else{
    return false;
}
    }
};
