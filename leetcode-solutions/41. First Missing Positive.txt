class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n=nums.size();
        int a=0;
          
        for(int i=0;i<n;i++){
            while(nums[i]>0 && nums[i]<=n && nums[i]!=nums[nums[i]-1]){
                int j=nums[i]-1;
                swap(nums[i],nums[j]);
            }
        }
        for(int i=0;i<n;i++){
            if(i+1!=nums[i]){
                a= i+1;
                break;
            }
            else{
                a=n+1;
            }
        }
        return a;
    }
};
