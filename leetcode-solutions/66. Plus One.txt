class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        int n=digits.size();
        vector<int>result;

        for(int i=0;i<n;i++){
            if(i==n-1){
                if(digits[i]<9){
                    result.push_back(digits[i]+1);
                }
                else{
                    result.push_back(1);
                    result.push_back(0);
                }
            }
            if(i<n-1){
                result.push_back(digits[i]);
            }
        }
        return result;
        
    }
};
