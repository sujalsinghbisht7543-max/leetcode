class Solution {
public:
    int firstUniqChar(string s) {
        int arr[26];
        for(int i=0;i<26;i++){
            arr[i]=-1;
        }

        for(int i=0;i<s.size();i++){
            if(arr[s[i]-'a']==-1){
                arr[s[i]-'a']=i;
            }
            else{
                arr[s[i]-'a']=-2;
            }
        }
        int ans=INT_MAX;

        for(int i=0;i<26;i++){
            if(arr[i]>=0){
                ans=min(arr[i],ans);
            }
        }
        if(ans==INT_MAX){
            return -1;
        }
        return ans;
    }
};
