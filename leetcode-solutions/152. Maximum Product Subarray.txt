class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int n = nums.size();
        long long ans = LLONG_MIN;
        long long pre = 1;
        long long suf = 1;
        
        for(int i = 0; i < n; i++) {
            pre = pre * nums[i];
            suf = suf * nums[n - i - 1];
            
            ans = max(ans, max(pre, suf));
         
            if(pre == 0) pre = 1;
            if(suf == 0) suf = 1;
        }
        
        return ans;
    }
};
