class Solution {
public:
    int sumOfSquares(vector<int>& nums) {
        int n=nums.size();
        
        int ans=nums[0]*nums[0];
        for(int i=1;i<n;i++){
            int x=i+1;
            if(n%x==0){
                ans+=nums[i]*nums[i];
            }
        }
        return ans;
    }
};
