class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        int n=numRows;
        vector<vector<int>>result;

        for(int i=0;i<n;i++){
            vector<int>ans;
            long long a=1;
            for(int j=0;j<=i;j++){
                ans.push_back(a);
                a=a*(i-j);
                a=a/(j+1);
            }
            result.push_back(ans);
        }
        return result;
        
    }
};
