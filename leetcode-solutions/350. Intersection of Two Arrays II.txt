class Solution {
public:
    vector<int> intersect(vector<int>& nums1, vector<int>& nums2) {
        multiset<int> a={nums1.begin(),nums1.end()};
        multiset<int> b={nums2.begin(),nums2.end()};

        vector<int> ans;

        for(int i : a){
            if(b.count(i)){
                ans.push_back(i);
               auto it= b.find(i);
                    b.erase(it);
                }
            }
        return ans;
    }
};
