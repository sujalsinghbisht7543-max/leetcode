class Solution {
public:
    int minEatingSpeed(vector<int>& piles, int h) {
        int n=piles.size();
        long long low=1;
        long long high=*max_element(piles.begin(),piles.end());

        while(low<=high){
            long long  k=(low+high)/2;
            long long step=0;
            for(int i=0;i<n;i++){
                if(piles[i]%k==0){
                    step+=piles[i]/k;
                }
                else{
                    step+=(piles[i]/k)+1;
                }
            }
            if(h<step){
                low=k+1;
            }
            else{
                high=k-1;
            }
        }
        return low;
    }
};
