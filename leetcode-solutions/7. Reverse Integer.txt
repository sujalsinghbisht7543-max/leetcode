class Solution {
public:
    int reverse(int x) {
        int m=INT_MAX;
        int n=INT_MIN;
        long long ans=0;
        int rem=0;
        while(x>0){
        rem=x%10;
        ans=(ans*10)+rem;
        x=x/10;
        if(ans>m){
            return 0;
        }
        }

        while(x!=0){
            rem=x%10;
            ans=(ans*10)+rem;
            x=x/10;   
        if(ans<n){
            return 0;
        }
        }


      return (int)ans;  
    }
};
