class Solution {
public:
    int rob(vector<int>& nums) {

        vector<int>l;
        vector<int>r;
        int n=nums.size();

        int e=0;
        int o=1;

        for(int i=0;i<n;i++){
            if(i%2==0){
                l.push_back(nums[e]);
                e+=2;
            }
            else{
                r.push_back(nums[o]);
                o+=2;
            }
        }
        int sum1=0;
        for(int i=0;i<l.size();i++){
            sum1+=l[i];
        }

         int sum2=0;
         for(int i=0;i<r.size();i++){
            sum2+=r[i];
         }

         int a=max(sum1,sum2);

       return a; 
    }
};
