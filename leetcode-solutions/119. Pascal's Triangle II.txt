class Solution {
public:
    vector<int> getRow(int rowIndex) {
        int row=rowIndex;
        vector<int>result;
        long long ans=1;
        result.push_back(ans);
        for(int i=0;i<row;i++){
            ans=ans*(row-i);
            ans=ans/(i+1);
            result.push_back(ans);
        }
        return result; 
    }
};
