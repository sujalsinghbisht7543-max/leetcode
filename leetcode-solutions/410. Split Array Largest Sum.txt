class Solution {
public:
    int splitArray(vector<int>& nums, int k) {
        int low=*max_element(nums.begin(),nums.end());
        int n=nums.size();
        int high=0;
        for(int i=0;i<n;i++){
            high+=nums[i];
        }
        while(low<=high){
            int mid=(low+high)/2;
            int sum=0;
            int count=0;
            for(int i=0;i<n;i++){
                sum+=nums[i];
                if(sum>mid){
                    count++;
                    sum=nums[i];
                }
                if(i==n-1){
                    count++;
                }
            }
        if(count>k){
            low=mid+1;
        }
        else{
            high=mid-1;
        }
        }
        return low;
    }
};
