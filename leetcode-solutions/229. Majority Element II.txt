class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
       int el1=-1;
       int el2=-1;
       int count1=0;
         vector<int>v;
       int count2=0;
       int n=nums.size();

       for(int i=0;i<n;i++){
        if(count1==0 && el2!=nums[i]){
            el1=nums[i];
            count1=1;
        }
        else if(count2==0 && el1!=nums[i]){
            el2=nums[i];
            count2=1;
        }
        else if(el1==nums[i]){
            count1++;
        }
        else if(el2==nums[i]){
            count2++;
        }
        else{
            count1--,count2--;
        }
       }

        int min=n/3;
        count1=0;
        count2=0;
       


        for(int i=0;i<n;i++){
            if(el1==nums[i]){
                count1++;
            }
           else if(el2==nums[i]){
                count2++;
            }
        }
        if(count1>n/3){
            v.push_back(el1);
        }
        if(count2>n/3){
            v.push_back(el2);
        }


       return v;

    }
};
