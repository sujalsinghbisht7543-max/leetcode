class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int n=nums.size();

        int low=0;
        int first=0;
        int high = n-1;
        vector<int>result;
        int ans=0;

        bool found=false;
        while(low<=high){
            int mid=(low+high)/2;

            if(nums[mid]==target){
                ans=mid;
                high=mid-1;
                found=true;
            }
            else if(nums[mid]<target){
                low=mid+1;
            }
            else{
                high=mid-1;
            }
        }
        if(found==false){
            result.push_back(-1);
            result.push_back(-1);
            return result;
        }
        else{
            result.push_back(ans);
            int low2=0;
        int high2 = n-1;
        int ans2=0;

        while(low2<=high2){
            int mid=(low2+high2)/2;

            if(nums[mid]==target){
                ans2=mid;
                low2=mid+1;
            }
            else if(nums[mid]<target){
                low2=mid+1;
            }
            else{
                high2=mid-1;
            }
        }
        result.push_back(ans2);
        }

        
        
        
        return result;
    }
};
