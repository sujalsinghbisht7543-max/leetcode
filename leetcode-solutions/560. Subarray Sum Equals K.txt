class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int,int>mp;
        mp[0]=1;
        int ps=0;
        int cnt=0;
        int n=nums.size();
        for(int i=0;i<n;i++){
            ps+=nums[i];

            int r=ps-k;
            cnt+=mp[r];
            mp[ps]+=1;
        }
        return cnt;
    }
};
