class Solution {
public:
    int findMin(vector<int>& nums) {
        int n=nums.size();
        int low=0;
        int high=n-1;
        int pivot=nums[0];

        while(low<=high){

            int mid=(low+high)/2;
            if(mid>0 && nums[mid]<nums[mid-1]){
                pivot=nums[mid];
                break;
            }
            else if(nums[mid]<nums[high]){
                high=mid-1;
            }
            else{
                low=mid+1;
            }
        
        }
        return pivot;
    }
};
