class Solution {
public:
    int alternatingSum(vector<int>& nums) {
        int n=nums.size();

        if(nums.size()==1){
            return nums[0];
        }

        int sum=0;
        int i=0;
        int j=1;
        
        while(i<n &&j<n){
            int minus=nums[i]-nums[j];
            sum+=minus;
            i+=2;
            j+=2;
        }
        while(i<n){
            sum+=nums[i];
            i+=2;
        }
        while(j<n){
            sum+=nums[j];
            j+=2;
        }
       return sum; 
    }
};
