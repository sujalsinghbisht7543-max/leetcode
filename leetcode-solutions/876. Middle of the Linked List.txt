class Solution {
public:
    ListNode* middleNode(ListNode* head) {
        ListNode *slow=head;
        ListNode *fast=head;

        while(fast!=NULL && fast->next!=NULL){
            slow=slow->next;
            if(fast->next!=NULL){
            fast=fast->next->next;
            }
        }
        return slow;

    
    }
};
