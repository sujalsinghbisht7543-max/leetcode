class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        int mid=m;
        int right=n;
        int l=0;
        int r=0;
        vector<int>temp(m+n);
        int k=0;

        while(l<mid && r<right){
            if(nums1[l]<nums2[r]){
                temp[k++]=nums1[l++];
            }
            else{
                temp[k++]=nums2[r++];
            }
        }
        while(l<mid){
            temp[k++]=nums1[l++];
        }
        while(r<right){
            temp[k++]=nums2[r++];
        }
        for(int i=0;i<m+n;i++){
            nums1[i]=temp[i];
        }
        
    }
};
