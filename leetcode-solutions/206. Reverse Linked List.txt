// using recursion
class Solution {
public:
ListNode* reverse(ListNode* head){
    if(head==NULL || head->next==NULL){
        return head;
    }

    ListNode* nhead=reverse(head->next);
    ListNode* front=head->next;
    front->next=head;
    head->next=NULL;
    return nhead;

}

    ListNode* reverseList(ListNode* head) {
         head=reverse(head);
        return head;
    }
};
