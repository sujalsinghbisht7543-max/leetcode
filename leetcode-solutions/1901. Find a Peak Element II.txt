class Solution {
public:
int findidx(vector<int> mat , int n){
    int maxvalue=-1;
    int maxidx=-1;
    for(int i=0;i<n;i++){
        if(mat[i]>maxvalue){
            maxvalue=mat[i];
            maxidx=i;
        }
    }
    return maxidx;
}
    vector<int> findPeakGrid(vector<vector<int>>& mat) {
        int m=mat.size();
        int n=mat[0].size();
         int low=0;
         int high=m-1;
         vector<int>ans;

         while(low<=high){
            int mid=(low+high)/2;

            int col=findidx(mat[mid],n);

            int up=mid-1<0 ? INT_MIN : mat[mid-1][col];
            int down=mid+1>=m ?INT_MIN : mat[mid+1][col];

            if(mat[mid][col]>up && mat[mid][col]>down){
                ans.push_back(mid);
                ans.push_back(col);
                break;
            }
            else if(mat[mid][col]<up){
                high=mid-1;
            }
            else{
                low=mid+1;
            }
         }
        return ans;
    }
};
